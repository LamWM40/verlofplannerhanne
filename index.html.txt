    <div id="overzicht"></div>

      <strong id="dagen">0.00 werkdagen</strong>

    <hr>
    <b>Gebruikte uren (S & B limiet)</b>
    <p id="resterend" style="text-align:right; font-weight:bold;"></p>
    <div class="bar-bg">
      <div class="bar-fill" id="bar"></div>
    </div>
    
    <button class="reset" onclick="resetData()">Alles Wissen</button>
  </div>

  <div class="card">
    <h3>Uren Ingeven</h3>
    <label>Soort verlof:</label>
    <select id="code">
      <option value="EL">EL (Extra Legaal)</option>
      <option value="R">R (Recup/Inhaalrust)</option>
      <option value="F">F (Feestdag)</option>
      <option value="C">C (Compensatie)</option>
      <option value="S">S (Senioriteit)</option>
      <option value="B">B (Basisverlof)</option>
    </select>
    
    <label>Aantal uren:</label>
    <input type="number" id="uren" placeholder="Bijv. 7.6">
    
    <label>Uren per werkdag:</label>
    <input type="number" id="urenPerDag" value="7.6" onchange="updateUrenPerDag()">
    
    <button onclick="toevoegen()">Toevoegen & Bereken</button>
  </div>

<script>
// Configuratie van de codes en hun vervaldata
const config = {
  'EL': { naam: 'Extra Legaal', vervalt: '31 dec 2026' },
  'R':  { naam: 'Inhaalrust (R)', vervalt: 'Binnen 3 maanden' },
  'F':  { naam: 'Vervangende Feestdag', vervalt: '31 dec 2026' },
  'C':  { naam: 'Compensatie', vervalt: '31 dec 2026' },
  'S':  { naam: 'Senioriteit', vervalt: '31 dec 2026', max: 15 },
  'B':  { naam: 'Basisverlof', vervalt: '31 dec 2026', max: 120 }
};

let data = JSON.parse(localStorage.getItem('verlof_data')) || {};
let urenPerDag = localStorage.getItem('uren_per_dag') || 7.6;
document.getElementById('urenPerDag').value = urenPerDag;

function toevoegen(){
  let code = document.getElementById('code').value;
  let uren = parseFloat(document.getElementById('uren').value);

  if(!uren || uren <= 0) return alert("Geef een geldig aantal uren in.");

  if(!data[code]) data[code] = 0;
  data[code] += uren;

  localStorage.setItem('verlof_data', JSON.stringify(data));
  document.getElementById('uren').value = ''; // input leegmaken
  toon();
}

function updateUrenPerDag() {
  urenPerDag = parseFloat(document.getElementById('urenPerDag').value);
  localStorage.setItem('uren_per_dag', urenPerDag);
  toon();
}

function resetData() {
  if(confirm("Weet je zeker dat je alle gegevens wilt wissen?")) {
    data = {};
    localStorage.removeItem('verlof_data');
    toon();
  }
}

function toon(){
  let overzichtHtml = '';
  let deadlineHtml = '';
  let totaalUren = 0;

  Object.keys(config).forEach(c => {
    let urenGeboekt = data[c] || 0;
    totaalUren += urenGeboekt;
    
    // Update Overzicht Kolom
    overzichtHtml += `
      <div class="info-row">
        <span><strong>${c}</strong> (${config[c].naam})</span>
        <span>${urenGeboekt}u</span>
      </div>`;

    // Update Deadline Kolom
    deadlineHtml += `
      <div class="info-row">
        <span>${config[c].naam}</span>
        <span class="deadline">${config[c].vervalt}</span>
      </div>`;
  });

  document.getElementById('overzicht').innerHTML = overzichtHtml;
  document.getElementById('deadlines').innerHTML = deadlineHtml;

  // Berekening dagen
  let dagen = totaalUren / urenPerDag;
  document.getElementById('dagen').innerText = `Totaal: ${dagen.toFixed(2)} werkdagen`;

  // Voortgangsbalk (S + B als voorbeeld van potje dat leeg moet)
  let used = (data['S'] || 0) + (data['B'] || 0);
  let totalPot = config.S.max + config.B.max;
  let percent = (used / totalPot) * 100;

  document.getElementById('resterend').innerText = `${used}u van ${totalPot}u verbruikt`;
  document.getElementById('bar').style.width = Math.min(percent, 100) + "%";
}

// Initialisatie
toon();
</script>
