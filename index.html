import json import socket import threading import time from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer from urllib.parse import urlparse # Pas dit aan naar het IP-adres van je batterij. BATTERY_IP = "192.168.0.107" BATTERY_PORT = 30000 WEB_PORT = 8080 POLL_INTERVAL_SECONDS = 5 UDP_TIMEOUT_SECONDS = 3 MAX_RESPONSE_BYTES = 4096 # LET OP: dit is een voorbeeldcommando; controleer welk commando # jouw Marstek-model en firmware daadwerkelijk ondersteunen. REQUEST_PAYLOAD = json.dumps({"cmd": "get_status"}).encode("utf-8") latest_data = {"status": "Verbinding zoeken..."} data_lock = threading.Lock() def set_latest_data(data): global latest_data with data_lock: latest_data = data def get_latest_data(): with data_lock: return latest_data.copy() def fetch_battery_data(): while True: try: with socket.socket(socket.AF_INET, socket.SOCK_DGRAM) as sock: sock.settimeout(UDP_TIMEOUT_SECONDS) sock.sendto(REQUEST_PAYLOAD, (BATTERY_IP, BATTERY_PORT)) response_bytes, address = sock.recvfrom(MAX_RESPONSE_BYTES) response_text = response_bytes.decode("utf-8").strip() parsed_data = json.loads(response_text) if not isinstance(parsed_data, dict): raise ValueError("De batterij stuurde JSON terug, maar geen JSON-object.") parsed_data.setdefault("status", "Verbonden") set_latest_data(parsed_data) print(f"Antwoord ontvangen van {address[0]}.") except socket.timeout: message = ( f"Geen antwoord van {BATTERY_IP}:{BATTERY_PORT} " f"binnen {UDP_TIMEOUT_SECONDS} seconden." ) set_latest_data({"status": "Offline / Geen antwoord", "error": message}) print(message) except json.JSONDecodeError as error: message = f"Antwoord van de batterij is geen geldige JSON: {error}" set_latest_data({"status": "Onverwacht antwoord", "error": message}) print(message) except OSError as error: message = f"Netwerkfout: {error}" set_latest_data({"status": "Netwerkfout", "error": message}) print(message) except Exception as error: message = f"Fout bij ophalen of verwerken van batterijdata: {error}" set_latest_data({"status": "Fout", "error": message}) print(message) time.sleep(POLL_INTERVAL_SECONDS) HTML_PAGE = """ Marstek Live Monitor

Marstek Live Monitor

Laadstatus (SoC)
--%
Vermogen (Watt)
-- W
Status
Verbinden...
""" class SimpleHandler(BaseHTTPRequestHandler): def do_GET(self): path = urlparse(self.path).path if path == "/data": body = json.dumps(get_latest_data()).encode("utf-8") self.send_response(200) self.send_header("Content-Type", "application/json; charset=utf-8") self.send_header("Cache-Control", "no-store") self.send_header("Content-Length", str(len(body))) self.end_headers() self.wfile.write(body) return body = HTML_PAGE.encode("utf-8") self.send_response(200) self.send_header("Content-Type", "text/html; charset=utf-8") self.send_header("Content-Length", str(len(body))) self.end_headers() self.wfile.write(body) def log_message(self, format_string, *args): # Houd de terminal rustig; verwijder deze methode om HTTP-requests te loggen. return if __name__ == "__main__": threading.Thread(target=fetch_battery_data, daemon=True).start() server = ThreadingHTTPServer(("0.0.0.0", WEB_PORT), SimpleHandler) print(f"Marstek-monitor gestart op http://localhost:{WEB_PORT}") print(f"Open op een ander apparaat in je netwerk: http://:{WEB_PORT}") try: server.serve_forever() except KeyboardInterrupt: print("\\nServer wordt afgesloten.") finally: server.server_close()
